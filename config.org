* System & Performance
Core system optimizations, garbage collection tweaks, and runtime file hygiene settings 
designed to keep Emacs fast, responsive, and uncluttered.

** Directory Cleanup (~no-littering~)
The ~no-littering~ package automatically redirects temporary and state files 
created by Emacs and its packages to the ~var/~ and ~etc/~ subdirectories inside =~/.emacs.d/=.

#+begin_src emacs-lisp
;; Garbage Collection & I/O Performance Optimizations
(setq gc-cons-threshold (* 50 1024 1024))
(setq read-process-output-max (* 3 1024 1024))

(use-package no-littering
  :ensure t
  :demand t
  :config
  ;; Redirect auto-save files (#filename#) to var/auto-save/
  (setq auto-save-file-name-transforms
        `((".*" ,(no-littering-expand-var-file-name "auto-save/") t)))

  ;; Redirect backup files (filename~) to var/backup/
  (setq backup-directory-alist
        `((".*" . ,(no-littering-expand-var-file-name "backup/")))))
#+end_src

** Smart Source Block Management & Auto-REPL
Handles safe creation, background REPL initialization, and smart buffer evaluation 
for Org source blocks across multiple languages (Racket, Python, Elisp).

*** 1. Safe Source Block Insertion 
#+begin_src emacs-lisp
(defun my/org-insert-src-block (lang)
  "Prompt for a LANG, insert a clean block, and safely open it."
  (interactive
   (list (completing-read "Language: "
                          '("racket" "python" "emacs-lisp" "scheme"))))
  
  ;; Use string= for string comparison
  (if (string= lang "racket")
      (insert (format "#+begin_src %s\n#lang racket\n\n#+end_src" lang))
    (insert (format "#+begin_src %s\n\n#+end_src" lang)))
  
  (forward-line -1)
  (org-edit-special))

(with-eval-after-load 'org
  (define-key org-mode-map (kbd "s-i") #'my/org-insert-src-block))
#+end_src

*** 2. Manual & Silent Background REPL Starter
#+begin_src emacs-lisp
;; Global option to prevent Geiser from popping up the REPL window
(setq geiser-repl-pop-to-buffer-p nil) 
;; Force Geiser to use Racket by default
(setq geiser-default-implementation 'racket)

(defun my/start-repl-background ()
  "Start the REPL silently in the background without modifying windows."
  (interactive)
  (cond
   ((derived-mode-p 'racket-mode)
    (save-window-excursion
      (run-geiser 'racket))
    (message "🚀 Racket REPL ready in the background."))

   ((derived-mode-p 'python-mode)
    (unless (python-shell-get-process)
      (save-window-excursion
        (run-python (python-shell-parse-command) nil nil)))
    (message "🚀 Python REPL ready in the background."))

   (t (message "No REPL configured for this language."))))

;; Bind to C-c r inside the C-c ' edit buffer
(with-eval-after-load 'org-src
  (define-key org-src-mode-map (kbd "C-c r") #'my/start-repl-background))
#+end_src

*** 3. Smart Buffer Evaluation (Run File)
#+begin_src emacs-lisp
(defun my/org-src-eval-buffer ()
  "Evaluate the entire source buffer dynamically depending on the language."
  (interactive)
  (cond
   ((derived-mode-p 'racket-mode)
    (geiser-mode 1) ;; Wake up Geiser if it was inactive
    (geiser-eval-buffer)
    (message "🚀 Racket: Full file evaluated in Geiser."))
   
   ((derived-mode-p 'python-mode)
    (python-shell-send-buffer)
    (message "🚀 Python: Full file evaluated."))
   
   ((derived-mode-p 'emacs-lisp-mode)
    (eval-buffer)
    (message "🚀 Elisp: Full file evaluated."))
   
   (t (message "No automatic evaluation defined for this mode."))))

;; Bind C-c C-b to our custom evaluator inside the edit window
(with-eval-after-load 'org-src
  (define-key org-src-mode-map (kbd "C-c C-b") #'my/org-src-eval-buffer))

** Smart Source Block Editing
Automatically opens `org-edit-special` when pressing RET on a source block header.

#+begin_src emacs-lisp
  (defun my/org-smart-dwim (orig-fun &rest args)
    "If cursor is on a src block header, open it. Otherwise, execute standard RET."
    (interactive "P")
    (let ((element (org-element-at-point)))
      (if (and (eq (org-element-type element) 'src-block)
               (= (line-number-at-pos)
                  (line-number-at-pos (org-element-property :post-affiliated element))))
          (org-edit-special)
        (apply orig-fun args))))

  ;; Advise org-return to use our smart behavior
  (with-eval-after-load 'org
    (advice-add 'org-return :around #'my/org-smart-dwim))
#+end_src

** Lisp Shared

#+begin_src emacs-lisp
(defun my/backward-transpose-sexp ()
  (interactive)
  (sp-transpose-sexp -1))

(defun my/kill-whitespace ()
  (interactive)
  (let ((start (point)))
    (skip-chars-forward " \t\n\r\f")
    (when (> (point) start)
      (delete-region start (point))
      t)))

(defun my/kill-whitespace-backward ()
  (interactive)
  (let ((start (point)))
    (skip-chars-backward " \t\n\r\f")
    (when (< (point) start)
      (delete-region (point) start)
      t)))

(defun my/kill-whitespace-or-sexp ()
  (interactive)
